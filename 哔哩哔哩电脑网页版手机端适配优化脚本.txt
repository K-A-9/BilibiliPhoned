            if (/\/v\/popular/.test(p)) return true;

        return false;

    const PAGE_CLASS_RULES = [
        { test: isMusicPage,    cls: ['bili-music-page'] },
        { test: isSearchPage,   cls: ['bili-search-page'] },
        { test: isTBiliDynamic, cls: ['bili-dynamic-page'] },
        { test: isOpusPage,     cls: ['bili-opus-page'] },
        { test: isHome,         cls: ['bili-home-page'] },
        { test: isSpacePage,    cls: ['bili-space-page'] },
        { test: isListPage,     cls: ['bili-video-page', 'bili-list-page'] },
        { test: isVideoPage,    cls: ['bili-video-page'] },
        { test: isBangumiPage,  cls: ['bili-bangumi-page'] },
        { test: isLiveRoomPage, cls: ['bili-live-room-page'] }
    ];
    const ALL_PAGE_CLASSES = [
        'bili-music-page', 'bili-search-page', 'bili-dynamic-page', 'bili-opus-page', 'bili-home-page',
        'bili-space-page', 'bili-video-page', 'bili-list-page', 'bili-bangumi-page', 'bili-live-room-page'
    ];

    function currentPageClasses() {
        for (let i = 0; i < PAGE_CLASS_RULES.length; i++) {
            if (PAGE_CLASS_RULES[i].test()) return PAGE_CLASS_RULES[i].cls;
        }
        return [];
    }
    function syncPageClass() {
        if (!document.body) return false;
        const want = currentPageClasses();
        const cl = document.body.classList;
        ALL_PAGE_CLASSES.forEach(c => { if (want.indexOf(c) === -1) cl.remove(c); });
        want.forEach(c => cl.add(c));
        if (isWideLayoutPage()) cl.add('bili-wide-page');
        else cl.remove('bili-wide-page');
        return true;
    }
    function applyPageClassEarly() {
        if (!syncPageClass()) {
            const obs = new MutationObserver(() => { if (syncPageClass()) obs.disconnect(); });
            obs.observe(document.documentElement, { childList: true, subtree: true });
        }
        window.addEventListener('popstate', syncPageClass);
        ['pushState', 'replaceState'].forEach(fn => {
            const orig = history[fn];
            if (typeof orig !== 'function') return;
            history[fn] = function () {
                const ret = orig.apply(this, arguments);
                setTimeout(syncPageClass, 0);
                return ret;
            };
        });
    }

    function fixViewport() {
        const apply = () => {
            const root = document.head || document.documentElement;
            if (!root) return;
            let vp = document.querySelector('meta[name="viewport"]');
            if (!vp) {
                vp = document.createElement('meta');
                vp.name = 'viewport';
                root.appendChild(vp);
            }
            const content = 'width=device-width, initial-scale=1.0, maximum-scale=1.0, user-scalable=no, viewport-fit=cover';
            if (vp.getAttribute('content') !== content) vp.setAttribute('content', content);
        };
        apply();
        [0, 50, 200, 500, 1000, 2000, 5000].forEach(t => setTimeout(apply, t));
        document.addEventListener('DOMContentLoaded', apply);
        window.addEventListener('load', apply);
    }

    function fixVideoControls() {
        if (!isVideoPage() && !isBangumiPage() && !isListPage()) return;
        if (fixVideoControls._done) return;

        const playerContainer = document.querySelector('body.bili-video-page .bpx-player-container') ||
                                 document.querySelector('body.bili-bangumi-page .bpx-player-container');
        if (!playerContainer) return;
        const videoArea = playerContainer.querySelector('.bpx-player-video-area');
        if (!videoArea) return;
        const videoPerch = videoArea.querySelector('.bpx-player-video-perch');
        const videoWrap = (videoPerch && videoPerch.querySelector('.bpx-player-video-wrap'))
                        || videoArea.querySelector('.bpx-player-video-wrap');
        if (!videoWrap) return;
        const video = videoWrap.querySelector('video');
        if (!video) return;

        if (videoWrap.parentElement === videoArea && videoArea.firstElementChild === videoWrap) {
            fixVideoControls._done = true;
            return;
        }
        if (videoPerch && videoWrap.parentElement !== videoArea) {
            videoArea.insertBefore(videoWrap, videoPerch);
        }
        video.playsInline = true;

        const oldControlWrap = videoArea.querySelector('.bpx-player-control-wrap:not(.new)');
        if (!oldControlWrap) return;
        const controlEntity = oldControlWrap.querySelector('.bpx-player-control-entity');
        if (!controlEntity) return;

    function bindGlobalWheelScroll() {
        if (bindGlobalWheelScroll._done) return;
        bindGlobalWheelScroll._done = true;

        document.addEventListener('wheel', function (e) {
            if (e.ctrlKey || e.metaKey) return;
            if (e.defaultPrevented) return;
            const t = e.target;
            if (!t || !t.closest) return;

            if (t.closest('#my-bili-bottom-nav, #my-bili-searchbar, #my-bili-mine-panel, #my-bili-dyn-upbar')) return;

            let el = t;
            while (el && el !== document.body && el !== document.documentElement) {
                let cs = null;
                try { cs = window.getComputedStyle(el); } catch (err) {}
                if (cs) {
                    const oy = cs.overflowY;
                    if ((oy === 'auto' || oy === 'scroll') && el.scrollHeight > el.clientHeight + 2) {
                        return;
                    }
                }
                el = el.parentElement;
            }

            let d = e.deltaY;
            if (e.deltaMode === 1) d *= 16;
            else if (e.deltaMode === 2) d *= Math.max(1, window.innerHeight);

            const scroller = document.scrollingElement || document.documentElement;
            const before = scroller.scrollTop;
            scroller.scrollTop = before + d;
            if (scroller.scrollTop !== before) e.preventDefault();
        }, { passive: false, capture: true });
    }

    function bindPanelMouseDrag() {
        if (bindPanelMouseDrag._done) return;
        bindPanelMouseDrag._done = true;

        let armed = false, sx = 0, sy = 0, st = 0;

        const getPanel = function () {
            if (isListPage()) return document.querySelector('body.bili-list-page .playlist-container--right');
            if (isVideoPage()) return document.querySelector('body.bili-video-page .right-container');
            if (isBangumiPage()) return document.querySelector('body.bili-bangumi-page .plp-r-wrap');
            return null;
        };
        const isOpen = function () {
            const r = getPanel();
            return !!(r && r.classList.contains('show'));
        };

        document.addEventListener('mousedown', function (e) {
            if (e.button !== 0) return;
            if (!isVideoPage() && !isBangumiPage() && !isListPage()) return;
            const t = e.target;
            if (!t || !t.closest) return;
            const panel = getPanel();
            if (panel && panel.contains(t)) return;
            const inPlayer = !!(t.closest('#playerWrap, #bilibili-player-wrap, .playlist-container--left, #mirror-vdcon'));
            const fromEdge = e.clientX > window.innerWidth - 32;
            if (!inPlayer && !fromEdge && !isOpen()) return;
            armed = true;
            sx = e.clientX; sy = e.clientY; st = Date.now();
        }, true);

        document.addEventListener('mouseup', function (e) {
            if (!armed) return;
            armed = false;
            if (e.button !== 0) return;
            if (Date.now() - st > 900) return;
            const dx = e.clientX - sx;
            const dy = e.clientY - sy;
            if (Math.abs(dx) < 60) return;
            if (Math.abs(dy) > Math.abs(dx)) return;
            const panel = getPanel();
            if (!panel) return;
            const mask = document.getElementById('my-bili-right-mask');
            if (dx < 0 && !isOpen()) { panel.classList.add('show'); if (mask) mask.classList.add('show'); }
            else if (dx > 0 && isOpen()) { panel.classList.remove('show'); if (mask) mask.classList.remove('show'); }
        }, true);
    }

    function getMyMid() {
        try {
            const m = String(document.cookie || '').match(/(?:^|;\s*)DedeUserID=(\d+)/);
            if (m && m[1]) return m[1];
        } catch (e) {}
        try {
            const s = window.__INITIAL_STATE__;

                const list = [s.mid, s.userMid, s.profile && s.profile.mid, s.userInfo && s.userInfo.mid, s.user && s.user.mid];
                for (let i = 0; i < list.length; i++) if (list[i]) return String(list[i]);
            }
        } catch (e) {}
        return '';
    }

    /* ============================================================
       搜索栏
       ============================================================ */
    const SEARCH_TYPES = [
        { key: 'all',     name: '综合', path: '/all' },
        { key: 'live',    name: '直播', path: '/live' },
        { key: 'article', name: '专栏', path: '/article' },
        { key: 'upuser',  name: '用户', path: '/upuser' }
    ];

    function currentSearchType() {
        const p = location.pathname.replace(/\/$/, '') || '/';
        let t = 'all';
        SEARCH_TYPES.forEach(function (x) { if (p === x.path) t = x.key; });
        return t;
    }

    function createSearchBar() {
        if (!isHome() && !isSearchPage() && !isSpacePage()) return;
        if (createSearchBar._started) return;
        createSearchBar._started = true;
        if (document.getElementById('my-bili-searchbar')) return;

        const onSearch = isSearchPage();
        const onSpace = isSpacePage();
        const url = new URL(location.href);
        const curType = currentSearchType();
        const onUpuser = onSearch && curType === 'upuser';
        const bar = document.createElement('div');
        bar.id = 'my-bili-searchbar';
        const showLeftBtn = !onSpace && !onUpuser;

        let leftBtnHTML = '';
        if (showLeftBtn) {
            if (onSearch) {
                leftBtnHTML = '<label id="my-bili-order-btn" for="my-bili-order-cb" role="button" tabindex="0" aria-label="排序">' +
                    '<svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round"><path d="M3 6h18"/><path d="M6 12h12"/><path d="M10 18h4"/></svg></label>';
            } else {
                leftBtnHTML = '<a id="my-bili-dynamic-btn" href="https://t.bilibili.com/" aria-label="动态">' +
                    '<svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round"><path d="M13 2L3 14h9l-1 8 10-12h-9l1-8z"/></svg></a>';
            }
        }
        let switchBtnHTML = '';
        let menuHTML = '';
        if (onSearch) {
            switchBtnHTML = '<label id="my-bili-switch-btn" for="my-bili-switch-cb" role="button" tabindex="0" aria-label="切换搜索类型">' +
                '<svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round">' +
                '<rect x="3" y="3" width="7" height="7" rx="1.5"/><rect x="14" y="3" width="7" height="7" rx="1.5"/>' +
                '<rect x="3" y="14" width="7" height="7" rx="1.5"/><rect x="14" y="14" width="7" height="7" rx="1.5"/></svg></label>';

            const orderItems = curType === 'live'
                ? [{ v: '', n: '综合' }, { v: 'live_time', n: '最新' }]
                : [{ v: '', n: '综合' }, { v: 'pubdate', n: '最新' }];
            const currentOrder = url.searchParams.get('order') || '';
            const isLatest = curType === 'live' ? (currentOrder === 'live_time') : (currentOrder === 'pubdate');
            menuHTML = (onUpuser ? '' :
                '<div class="my-bili-menu-row" id="my-bili-order-row">' +
                    orderItems.map(function (o) {
                        return '<a class="my-bili-menu-item' + (!!o.v === isLatest ? ' is-current' : '') + '" href="#" data-order="' + o.v + '">' + o.n + '</a>';
                    }).join('') +
                '</div>') +
                '<div class="my-bili-menu-row" id="my-bili-switch-row">' +
                    SEARCH_TYPES.map(function (t) {
                        return '<a class="my-bili-menu-item' + (t.key === curType ? ' is-current' : '') + '" href="#" data-type="' + t.key + '">' + t.name + '</a>';
                    }).join('') +
                '</div>';
        }
        const cbHTML = !onSearch ? '' :
            (onUpuser ? '' : '<input type="checkbox" id="my-bili-order-cb" class="my-bili-menu-cb">') +
            '<input type="checkbox" id="my-bili-switch-cb" class="my-bili-menu-cb">';

        const placeholderText = onSpace ? '搜索视频、动态'
            : '搜索视频、番剧、UP主...（用 -词 排除）';
        const kw = onSearch ? (url.searchParams.get('keyword') || '') : '';
        const escapedKw = kw.replace(/"/g, '&quot;').replace(/</g, '&lt;').replace(/>/g, '&gt;');

        bar.innerHTML = cbHTML + leftBtnHTML +
            '<div class="my-search-wrap">' +
                '<input type="search" id="my-bili-search-input" placeholder="' + placeholderText + '" autocomplete="off" value="' + escapedKw + '" />' +
                '<button class="my-search-btn" aria-label="搜索">' +
                    '<svg width="18" height="18" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2.2" stroke-linecap="round"><circle cx="11" cy="11" r="7"></circle><line x1="16.5" y1="16.5" x2="21" y2="21"></line></svg>' +
                '</button>' +
            '</div>' +
            switchBtnHTML + menuHTML;

        const attach = () => {
            if (!document.body) { setTimeout(attach, 50); return; }
            Array.prototype.slice.call(document.querySelectorAll('#my-bili-searchbar'))
                .forEach(function (el) { if (el !== bar && el.parentNode) el.parentNode.removeChild(el); });
            document.body.appendChild(bar);

            const input = bar.querySelector('#my-bili-search-input');
            const btn = bar.querySelector('.my-search-btn');

            const parseQueryTokens = function (raw) {
                const tokens = String(raw || '').split(/\s+/).filter(Boolean);
                const include = [], exclude = [];
                tokens.forEach(function (t) {
                    if (t.length > 1 && t.charAt(0) === '-') exclude.push(t.slice(1));
                    else if (t.length > 1 && t.charAt(0) === '+') include.push(t.slice(1));
                    else include.push(t);
                });
                return { include: include, exclude: exclude };
            };

            const doSearch = () => {
                const parsed = parseQueryTokens(input.value.trim());
                const keyword = parsed.include.join(' ');
                if (!keyword) return;
                if (onSpace) {
                    const mid = location.pathname.split('/').filter(Boolean)[0] || '';
                    if (!mid || !/^\d+$/.test(mid)) return;
                    location.href = `https://space.bilibili.com/${mid}/search?keyword=${encodeURIComponent(keyword)}`;
                    return;
                }
                const ct = currentSearchType();
                let base = 'https://search.bilibili.com/all';
                SEARCH_TYPES.forEach(function (t) { if (t.key === ct) base = 'https://search.bilibili.com' + t.path; });
                const next = new URL(base);
                next.searchParams.set('keyword', keyword);
                if (ct !== 'all') {
                    next.searchParams.set('vt', '63088886');
                    next.searchParams.set('from_source', 'web_search');
                }
                if (ct === 'live') next.searchParams.set('search_type', 'live_room');
                location.href = next.toString();
            };
            input.addEventListener('keydown', e => {
                if (e.key === 'Enter' || e.keyCode === 13) { e.preventDefault(); doSearch(); }
            });
            btn.addEventListener('click', doSearch);

            const allCb = () => bar.querySelectorAll('.my-bili-menu-cb');
            const clearMarks = () => {
                bar.querySelectorAll('.my-bili-menu-row').forEach(function (r) { r.classList.remove('is-open'); });
                bar.querySelectorAll('#my-bili-order-btn, #my-bili-switch-btn').forEach(function (b) { b.classList.remove('is-open'); });
            };
            const closeAll = () => { allCb().forEach(function (o) { o.checked = false; }); clearMarks(); };
            const setOpen = function (cb, open) {
                allCb().forEach(function (o) { o.checked = false; });
                clearMarks();
                cb.checked = !!open;
                if (!open || !cb.id) return;
                const base = cb.id.replace(/-cb$/, '');
                const row = bar.querySelector('#' + base + '-row');
                const btn = bar.querySelector('#' + base + '-btn');
                if (row) row.classList.add('is-open');
                if (btn) btn.classList.add('is-open');
            };
            bar.querySelectorAll('label[for]').forEach(function (lb) {
                lb.addEventListener('click', function (e) {
                    const cb = bar.querySelector('#' + lb.getAttribute('for'));
                    if (!cb) return;
                    if (e && typeof e.preventDefault === 'function') e.preventDefault();
                    setOpen(cb, !cb.checked);
                });
            });
            document.addEventListener('click', function (e) {
                if (bar.contains(e.target)) return;
                closeAll();
            });
            bar.querySelectorAll('.my-bili-menu-item').forEach(function (a) {
                a.addEventListener('click', function () { setTimeout(closeAll, 0); });
            });
        };
        attach();
    }

    function hideWatchLaterButtons() {
        if (!document.querySelectorAll) return;
        const all = document.querySelectorAll('body *');
        for (let i = 0; i < all.length; i++) {
            const el = all[i];
            if (!el.classList) continue;
            if (el.classList.contains('bili-hide-watch-later')) continue;
            let hit = false;
            const cls = el.className || '';
            const clsStr = typeof cls === 'string' ? cls : String(cls || '');
            let aria = '';
            try {
                aria = (el.getAttribute && (el.getAttribute('aria-label') || '')) + ' ' +
                       (el.getAttribute && (el.getAttribute('title') || ''));
            } catch (e) { aria = ''; }
            const hay = (clsStr + ' ' + aria).toLowerCase();
            ['watch-later', 'watchlater', '稍后再看'].forEach(function (k) {
                if (hay.indexOf(k) !== -1) hit = true;
            });
            if (!hit) {
                const txt = ((el.textContent || '') + '').trim();
                if (txt !== '稍后再看') continue;
            }
            let w = 0, h = 0;
            try {
                const r = el.getBoundingClientRect();
                w = r.width; h = r.height;
            } catch (e) { continue; }
            if (w > 120 || h > 120) continue;
            if (w === 0 && h === 0) continue;
            el.classList.add('bili-hide-watch-later');
        }
    }

    function fixSpaceNavTabs() {
        if (!isSpacePage()) return;
        const allowedTabs = ['主页', '动态', '投稿', '合集和系列'];
        document.querySelectorAll('body.bili-space-page .nav-tab > a.nav-tab__item').forEach(el => {
            const text = (el.textContent || '').trim();
            el.style.removeProperty('display');
            el.style.removeProperty('visibility');
            const isAllowed = allowedTabs.some(t => text.indexOf(t) !== -1);
            if (!isAllowed) {
                el.style.setProperty('display', 'none', 'important');
                el.style.setProperty('visibility', 'hidden', 'important');
            }
        });
    }

    /* ============================================================
       run
       ============================================================ */
    function run() {
        if (!document.body) return;
        syncPageClass();
        createSearchBar();
        createBottomNav();
        syncBottomNavActive();
        hideWatchLaterButtons();

        if (isTBiliDynamic()) {
            createDynamicUpBar();
        } else if (isOpusPage()) {
            /* opus */
        } else if (isHome()) {
            /* home */
        } else if (isSearchPage()) {
            /* search */
        } else if (isSpacePage()) {
            fixSpaceNavTabs();
        } else if (isVideoPage() || isListPage()) {
            fixVideoControls();
            bindVideoSlideGesture();
            bindFullscreenFix();
            bindPlayerHoverControl();
            disableMiniPlayer();
            if (isVideoPage()) fixUpAvatar();
        } else if (isBangumiPage()) {
            fixVideoControls();
            bindVideoSlideGesture();
            bindPlayerHoverControl();
            disableMiniPlayer();
        } else if (isLiveRoomPage()) {
            /* 直播间 */
        }
    }

    /* ============================================================
       启动
       ============================================================ */
    function start() {
        injectStyle();
        applyPageClassEarly();
        fixViewport();
        createSearchBar();
        bindGlobalWheelScroll();
        bindPanelMouseDrag();

        const runAll = () => {
            run();
            setTimeout(run, 300);
            setTimeout(run, 800);
            setTimeout(run, 1500);
            setTimeout(run, 3000);
            setTimeout(run, 5000);
            setTimeout(run, 8000);
        };

        if (document.readyState === 'loading') {
            document.addEventListener('DOMContentLoaded', runAll);
        } else {
            runAll();
        }
        window.addEventListener('load', runAll);

        const observer = new MutationObserver(() => {
            if (observer._t) clearTimeout(observer._t);
            observer._t = setTimeout(() => { run(); }, 200);
        });
        const observeStart = () => {
            if (!document.body) { setTimeout(observeStart, 50); return; }
            observer.observe(document.body, { childList: true, subtree: true });
        };
        observeStart();
    }

    start();
})();
